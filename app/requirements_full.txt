# 安裝步驟（必須照順序）：
#
#   1. 建立 venv（建議 Python 3.10–3.12）：
#      python -m venv venv
#
#   2. 先安裝目標 profile 的 PyTorch（CUDA / CPU / ROCm 必須使用不同環境）：
#      CPU 範例：
#      venv\Scripts\pip install torch torchaudio --index-url https://download.pytorch.org/whl/cpu
#
#   3. 安裝本檔所有依賴：
#      venv\Scripts\pip install -r requirements_full.txt
#
#   4. （建議）若你也要直接從原始碼執行 app，可另外安裝本地魔改版套件：
#      venv\Scripts\pip install -e ..\stream-translator-gpt
#
#   5. （可選）若要啟用 Qwen3-ASR 4-bit 量化：
#      venv\Scripts\pip install bitsandbytes
#
#   6. 執行目標 profile 打包：
#      .\build_release.ps1 -Profile cpu -Version 1.3.7
